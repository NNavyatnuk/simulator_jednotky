cmake_minimum_required(VERSION 3.16)
project(simulator_jednotky LANGUAGES CXX)

add_executable(simulator_jednotky src/main.cpp)
target_compile_features(simulator_jednotky PRIVATE cxx_std_17)
set_target_properties(simulator_jednotky PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(simulator_jednotky PRIVATE SIMULATOR_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(simulator_jednotky PRIVATE /W4 /utf-8)
else()
  target_compile_options(simulator_jednotky PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(simulator_jednotky PRIVATE ws2_32)
endif()
