cmake_minimum_required(VERSION 3.28)
project(simulator_jednotky VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)

option(SIMULATOR_BUILD_DESKTOP "Build the native Qt Widgets application" ON)
option(SIMULATOR_BUILD_WEB_SERVER "Build the legacy HTTP server (not a desktop GUI)" OFF)
include(CTest)

if(SIMULATOR_BUILD_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets SerialPort)
  include(FetchContent)
  # MIT-licensed QuickJS-ng; immutable contents verified on every fresh fetch.
  FetchContent_Declare(quickjs
    URL https://codeload.github.com/quickjs-ng/quickjs/tar.gz/refs/tags/v0.10.1
    URL_HASH SHA256=4e917eea3bb6051a0551fdd3aac45199ff705fa08920c15b824a08c5fbb032e4
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    EXCLUDE_FROM_ALL)
  set(QJS_BUILD_LIBC OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(quickjs)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_library(simulator_desktop_core STATIC desktop/engine.cpp desktop/resources.qrc desktop/serial.cpp)
  target_include_directories(simulator_desktop_core PUBLIC desktop)
  target_link_libraries(simulator_desktop_core PUBLIC Qt6::Widgets Qt6::SerialPort qjs)
  add_library(simulator_desktop_ui STATIC desktop/window.cpp)
  target_link_libraries(simulator_desktop_ui PUBLIC simulator_desktop_core)
  qt_add_executable(simulator_jednotky WIN32 desktop/main.cpp)
  target_link_libraries(simulator_jednotky PRIVATE simulator_desktop_ui)
  foreach(target simulator_desktop_core simulator_desktop_ui simulator_jednotky)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endforeach()
  if(MSVC)
    # Embedded JS uses a bounded 512 KB stack; leave room for Qt and C++ calls.
    target_link_options(simulator_jednotky PRIVATE /STACK:8388608)
  endif()
  # Keep the executable beside deployed Qt DLLs and qt.conf. Qt's deployment
  # API derives this location from CMAKE_INSTALL_BINDIR (bin by default).
  install(TARGETS simulator_jednotky RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  qt_generate_deploy_app_script(TARGET simulator_jednotky OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
  install(FILES ${quickjs_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME QuickJS-ng-LICENSE)
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(desktop_tests desktop/tests.cpp)
    target_link_libraries(desktop_tests PRIVATE simulator_desktop_ui Qt6::Test)
    if(MSVC)
      target_compile_options(desktop_tests PRIVATE /utf-8)
      target_link_options(desktop_tests PRIVATE /STACK:8388608)
    endif()
    add_test(NAME desktop COMMAND desktop_tests)
    add_test(NAME desktop_startup COMMAND simulator_jednotky --smoke-test)
    set_tests_properties(desktop desktop_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()

if(SIMULATOR_BUILD_WEB_SERVER)
  add_executable(simulator_http src/main.cpp)
  target_compile_definitions(simulator_http PRIVATE SIMULATOR_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(simulator_http PRIVATE /W4 /utf-8)
  else()
    target_compile_options(simulator_http PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    target_link_libraries(simulator_http PRIVATE ws2_32)
  endif()
endif()
